% ── Slide 2: Problema e Objetivo ──────────────────────────────────────────────
\begin{frame}{Introdução}

    \begin{itemize}
        \item[] Regime Próprio de Previdência Social (RPPS)
        \medskip
        \item[] Portaria MTP nº 1.467/2022
        \begin{itemize}
            \item Art. 36
        \end{itemize}
    \end{itemize}

    \begin{center}
    \begin{minipage}{0.55\textwidth}
        \begin{alertblock}{Hipótese}
        \[
            e_{\overline{X}}^{*} \geq e_{\overline{X}}^{r}
            \Rightarrow
            PM^{*} \geq PM^{r}
        \]
        \end{alertblock}
    \end{minipage}
    \end{center}
    \medskip
    \begin{itemize}
    \item[] A PM é uma \textbf{função da estrutura completa} de probabilidades de sobrevivência, não apenas de um ponto:
    \[ PM_i = f\!\left(\{_t p_x^{(i)}\}_{t \geq 0}\right) = VABF - VACF \]
    \end{itemize}


\end{frame}

% ── Slide 3: Por que a hipótese pode falhar? ──────────────────────────────────
\begin{frame}{Objetivo e justificativa}

    \begin{itemize}
        \item[] Objetivo:
        \begin{itemize}
            \item Verificar se alguma tábua comparativa resulta em uma PM inferior à tábua de referência, mesmo com expectativas de vida na idade média superior.
        \end{itemize}
        \medskip
        \item[] Justificativa:
        \begin{itemize}
            \item Expectativa de vida média abstrai a dispersão temporal das probabilidades de sobrevivência
            \item Rocha \textit{et al.} (2024).
        \end{itemize}
    \end{itemize}


\end{frame}

% ── Slide 4: Metodologia ──────────────────────────────────────────────────────
\begin{frame}{Metodologia}
  \begin{columns}[T]
    % Coluna esquerda
    \begin{column}{0.45\textwidth}

      \vspace{1.5cm}
      \begin{itemize}
        \item[] Massas simuladas (100.000 indivíduos cada)
        \begin{itemize}
          \item Base jovem: idade média $\approx$ 33 anos
          \item Base idosa: idade média $\approx$ 65 anos
        \end{itemize}
      \end{itemize}
      \smallskip
      \begin{itemize}
        \item[] Cenário
        \begin{itemize}
          \item Ambiente unidecremental
        \end{itemize}
      \end{itemize}
    \end{column}

    % Coluna direita — algoritmo como figura
    \begin{column}{0.52\textwidth}
      \begin{figure}
        \centering
        \scalebox{0.65}{%
          \begin{minipage}{\linewidth}
            \begin{algorithm}[H]
              \caption{Seleção de tábuas}
              \KwIn{Base de dados e conjunto de tábuas}
              \KwOut{Conjunto de tábuas aprovadas}
              Carregar base de dados\;
              Calcula-se a idade média geral do grupo\;
              A base é dividida em subconjuntos, um para cada gênero\;
              \Para{cada gênero}{
                Busca-se na tábua de referência (IBGE 2024) a expectativa de vida na idade média do grupo\;
                \Para{cada tábua comparativa}{
                  Busca-se a expectativa de vida na idade média do grupo\;
                  \eSe{$e_x^{\text{comp}} \geq e_x^{\text{ref}}$}{
                    Aprovar a tábua\;
                  }{
                    Descartar a tábua\;
                  }
                }
              }
            \end{algorithm}
          \end{minipage}
        }
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

% ── Slide 6: Conclusões ───────────────────────────────────────────────────────
\begin{frame}{Conclusões e Limitações}

  \begin{itemize}
    \item[] Conclusões 
    \begin{enumerate}
      \item O critério do Art.\,36 pode não ser suficiente.

      \medskip
      \item Estruturas etárias jovens, em especial a do sexo masculino, podem estar mais suscetíveis a essas tábuas.

      \medskip
      \item Fatores de suavização e agravamento.
    \end{enumerate}
    \medskip
    \item[] Limitações
    \begin{itemize}
      \item Uso exclusivo de dados simulados
      \medskip
      \item Simplificação do cenário proposto
    \end{itemize} 

  \end{itemize}
